    def copy_daily_events_table(self) -> None:
        table = self.daily_events_table

        selected_rows = sorted({
            index.row()
            for index in table.selectionModel().selectedRows()
        })

        if not selected_rows:
            selected_rows = list(range(table.rowCount()))

        headers = [
            table.horizontalHeaderItem(column).text()
            for column in range(table.columnCount())
        ]

        lines = ["\\t".join(headers)]

        for row in selected_rows:
            values = []
            for column in range(table.columnCount()):
                item = table.item(row, column)
                values.append(item.text() if item else "")
            lines.append("\\n".join(["\\t".join(values)]))

        QApplication.clipboard().setText("\\n".join(lines))

    def export_daily_events_table(self) -> None:
        table = self.daily_events_table

        file_path, _ = QFileDialog.getSaveFileName(
            self,
            "????? ????? ????????? ??????",
            "daily_events.csv",
            "CSV Files (*.csv)",
        )

        if not file_path:
            return

        import csv

        headers = [
            table.horizontalHeaderItem(column).text()
            for column in range(table.columnCount())
        ]

        with open(
            file_path,
            "w",
            encoding="utf-8-sig",
            newline="",
        ) as file:
            writer = csv.writer(file)
            writer.writerow(headers)

            for row in range(table.rowCount()):
                writer.writerow([
                    table.item(row, column).text()
                    if table.item(row, column)
                    else ""
                    for column in range(table.columnCount())
                ])

        # price chart
        series_data = m.price_series(conn, start, end)
        chart = QChart()
        chart.setBackgroundBrush(QColor("#FFFFFF"))
        chart.setTitle("روند قیمت روزانه سهم")

        line = QLineSeries()
        line.setName("بیشترین قیمت روزانه")

        categories = []
        valid_highs = []

        for i, d in enumerate(series_data):
            high = float(d["high"] or 0)
            line.append(float(i), high)
            valid_highs.append(high)
            g_date = dt.date.fromisoformat(d["date"])
            categories.append(to_jalali_str(g_date)[5:])

        chart.addSeries(line)
        # Tooltip for daily price points
        def show_price_tooltip(point, state):
            if not state:
                QToolTip.hideText()
                return

            index = int(round(point.x()))
            if 0 <= index < len(series_data):
                d = series_data[index]
                g_date = dt.date.fromisoformat(d["date"])
                jalali_date = to_jalali_str(g_date)

            # Use the actual chart-view position of the hovered point.
            chart_pos = self.price_chart_view.chart().mapToPosition(point, line)
            tooltip_pos = self.price_chart_view.mapToGlobal(
                chart_pos.toPoint()
            )

            QToolTip.showText(
                tooltip_pos,
                f"تاریخ: {jalali_date}\n"
                f"بیشترین قیمت روزانه: {float(d['high']):,.0f}",
                self.price_chart_view
            )

        line.hovered.connect(show_price_tooltip)
        # Numeric X axis — compatible with QLineSeries
        axis_x = QCategoryAxis()
        axis_x.setRange(0, max(1, len(categories) - 1))

        if categories:
            step = max(1, len(categories) // 10)
            for idx, label in enumerate(categories):
                if idx % step == 0 or idx == len(categories) - 1:
                    axis_x.append(label, float(idx))

        chart.addAxis(axis_x, Qt.AlignBottom)
        line.attachAxis(axis_x)

        # Y axis uses ONLY the actual daily highest prices
        axis_y = QValueAxis()
        if valid_highs:
            min_price = min(valid_highs)
            max_price = max(valid_highs)

            if min_price == max_price:
                margin = max(1.0, min_price * 0.02)
            else:
                margin = (max_price - min_price) * 0.05

            axis_y.setRange(
                max(0, min_price - margin),
                max_price + margin,
            )

        axis_y.setLabelFormat("%.0f")
        chart.addAxis(axis_y, Qt.AlignLeft)
        line.attachAxis(axis_y)

        chart.legend().hide()
